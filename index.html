import { serve } from "https://deno.land/std@0.224.0/http/server.ts"; import { createClient } from "https://esm.sh/@supabase/supabase-js@2"; const URL = Deno.env.get("SUPABASE_URL")!; const SERVICE_KEY = Deno.env.get("SUPABASE_SERVICE_ROLE_KEY")!; const db = createClient(URL, SERVICE_KEY, { auth: { persistSession: false } }); const MAX_FILE_SIZE = 10 * 1024 * 1024; const BUCKET = "st-payment-receipts"; const MIME: Record = { jpg: "image/jpeg", jpeg: "image/jpeg", png: "image/png", webp: "image/webp", heic: "image/heic", heif: "image/heif", dng: "image/x-adobe-dng" }; const cors = { "Access-Control-Allow-Origin": "*", "Access-Control-Allow-Methods": "POST, OPTIONS", "Access-Control-Allow-Headers": "authorization, apikey, content-type, x-client-info", "Cache-Control": "no-store" }; function reply( status: number, body: Record ) { return new Response(JSON.stringify(body), { status, headers: { ...cors, "Content-Type": "application/json; charset=utf-8" } }); } function fail(message: string, status = 400) { return reply(status, { success: false, error: message }); } function clean(value: FormDataEntryValue | null) { return typeof value === "string" ? value.trim() : ""; } async function hash(value: string) { const bytes = new TextEncoder().encode(value); const digest = await crypto.subtle.digest("SHA-256", bytes); return Array.from(new Uint8Array(digest)) .map(x => x.toString(16).padStart(2, "0")) .join(""); } async function checkLimit( key: string, max: number, seconds: number ) { const { data, error } = await db.rpc( "st_check_upload_limit", { p_key: key, p_max: max, p_window_seconds: seconds } ); if (error) { console.error("Rate limit database error:", error.code); throw new Error("تعذر التحقق من حدود التسجيل"); } return data === true; } async function validSignature( file: File, extension: string ) { const bytes = new Uint8Array( await file.slice(0, 32).arrayBuffer() ); if (extension === "jpg" || extension === "jpeg") { return bytes[0] === 0xff && bytes[1] === 0xd8 && bytes[2] === 0xff; } if (extension === "png") { return bytes[0] === 0x89 && bytes[1] === 0x50 && bytes[2] === 0x4e && bytes[3] === 0x47; } if (extension === "webp") { const s = new TextDecoder().decode(bytes); return s.startsWith("RIFF") && s.slice(8, 12) === "WEBP"; } if (extension === "heic" || extension === "heif") { const s = new TextDecoder().decode(bytes); const brand = s.slice(8, 12); return s.slice(4, 8) === "ftyp" && [ "heic", "heix", "hevc", "hevx", "mif1", "msf1", "heif" ].includes(brand); } if (extension === "dng") { const littleEndian = bytes[0] === 0x49 && bytes[1] === 0x49 && bytes[2] === 0x2a && bytes[3] === 0x00; const bigEndian = bytes[0] === 0x4d && bytes[1] === 0x4d && bytes[2] === 0x00 && bytes[3] === 0x2a; return littleEndian || bigEndian; } return false; } serve(async (req: Request) => { if (req.method === "OPTIONS") { return new Response(null, { status: 204, headers: cors }); } if (req.method !== "POST") { return fail("طريقة الطلب غير مسموحة", 405); } let uploadedPath: string | null = null; let registered = false; try { const length = Number( req.headers.get("content-length") || 0 ); if (length > MAX_FILE_SIZE + 1024 * 1024) { return fail("حجم الطلب أكبر من المسموح", 413); } const type = req.headers.get("content-type") || ""; if (!type.toLowerCase().includes("multipart/form-data")) { return fail("صيغة الطلب غير صحيحة", 415); } const form = await req.formData(); const tournamentId = clean(form.get("tournament_id")); const playerOne = clean(form.get("player_one")); const playerTwo = clean(form.get("player_two")); const phone = clean(form.get("phone")); const phoneTwo = clean(form.get("phone_two")); const reference = clean(form.get("reference")); const file = form.get("receipt"); if (!tournamentId || tournamentId.length > 120) { return fail("البطولة غير صحيحة"); } if ( playerOne.length < 2 || playerTwo.length < 2 || playerOne.length > 120 || playerTwo.length > 120 || playerOne.toLocaleLowerCase() === playerTwo.toLocaleLowerCase() ) { return fail("راجع أسماء اللاعبين"); } const phonePattern = /^01[0125][0-9]{8}$/; if ( !phonePattern.test(phone) || !phonePattern.test(phoneTwo) || phone === phoneTwo ) { return fail("راجع أرقام موبايل اللاعبين"); } if (reference.length > 100) { return fail("رقم مرجع العملية طويل"); } if (!(file instanceof File)) { return fail("صورة الإيصال مطلوبة"); } if ( file.size === 0 || file.size > MAX_FILE_SIZE ) { return fail("حجم الصورة يجب ألا يتجاوز 10 ميجابايت"); } const extension = file.name .split(".") .pop() ?.toLowerCase() || ""; const contentType = MIME[extension]; if (!contentType) { return fail("صيغة صورة الإيصال غير مدعومة"); } if (!(await validSignature(file, extension))) { return fail("ملف الإيصال لا يطابق صيغة الصورة"); } // Gateway-provided IP is a best-effort signal. const forwarded = req.headers.get("x-forwarded-for") || ""; const ip = forwarded.split(",")[0].trim() || req.headers.get("cf-connecting-ip") || "unknown"; const ipHash = await hash(ip); const phoneHash = await hash(phone); const secondPhoneHash = await hash(phoneTwo); // 5 attempts per hour per apparent IP. if (!(await checkLimit( "ip:" + ipHash, 5, 3600 ))) { return fail( "محاولات تسجيل كثيرة. حاول بعد ساعة", 429 ); } // 3 attempts per hour per player phone. if (!(await checkLimit( "phone:" + phoneHash, 3, 3600 ))) { return fail( "تم تجاوز عدد المحاولات لهذا الرقم", 429 ); } if (!(await checkLimit( "phone:" + secondPhoneHash, 3, 3600 ))) { return fail( "تم تجاوز عدد المحاولات لهذا الرقم", 429 ); } const path = crypto.randomUUID() + "." + extension; const { error: uploadError } = await db.storage .from(BUCKET) .upload(path, file, { contentType, upsert: false, cacheControl: "0" }); if (uploadError) { console.error("Upload failed:", uploadError.message); return fail("تعذر رفع صورة الإيصال", 500); } uploadedPath = path; const { data, error } = await db.rpc( "st_submit_registration_v2", { p_tournament_id: tournamentId, p_player_one: playerOne, p_player_two: playerTwo, p_phone: phone, p_phone_two: phoneTwo, p_receipt_path: path, p_reference: reference || null } ); if (error) { console.error( "Registration rejected:", error.code, error.message ); return fail( "تعذر تسجيل الطلب: " + error.message ); } registered = true; return reply(200, { success: true, registration_id: data, message: "تم إرسال الطلب للمراجعة" }); } catch (error) { console.error("Secure registration error:", error); return fail( "تعذر إتمام التسجيل. حاول مرة أخرى", 500 ); } finally { // Remove orphan receipt when registration fails. if (uploadedPath && !registered) { const { error } = await db.storage .from(BUCKET) .remove([uploadedPath]); if (error) { console.error( "Receipt cleanup failed:", error.message ); } } } });
